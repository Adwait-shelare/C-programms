#include <stdio.h>
int main(){
	int n,positivesum=0,negativesum=0;
	scanf("%d",&n);
	int arr[n];
	for(int i=0;i<n;i++){
		scanf("%d",&arr[i]);
	if(arr[i]>0){
		positivesum= positivesum+arr[i];
	}
	if(arr[i]<0){
		negativesum = negativesum+arr[i];
	}
	}
	printf("%d\n",positivesum);
	printf("%d\n",negativesum);
	
}
